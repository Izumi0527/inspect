#### 2.2 InfluxDB 完美契合方案

基于用户要求，我们需要优化 InfluxDB 的使用，确保性能、可靠性和可扩展性。

**2.2.1 连接管理优化**

```python
# backend/src/core/influxdb.py
from influxdb_client import InfluxDBClient
from influxdb_client.client.write_api import SYNCHRONOUS, ASYNCHRONOUS
import asyncio
from functools import lru_cache

class InfluxDBManager:
    """InfluxDB 连接管理器 - 单例模式"""

    _instance = None
    _client: Optional[InfluxDBClient] = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        if self._client is None:
            self._client = InfluxDBClient(
                url=settings.INFLUXDB_URL,
                token=settings.INFLUXDB_TOKEN,
                org=settings.INFLUXDB_ORG,
                timeout=10_000,  # 10秒超时
                enable_gzip=True,  # 启用压缩
            )

    @property
    def query_api(self):
        """查询 API（线程安全）"""
        return self._client.query_api()

    @property
    def write_api(self):
        """写入 API（异步批量写入）"""
        return self._client.write_api(write_options=ASYNCHRONOUS)

    async def health_check(self) -> bool:
        """健康检查"""
        try:
            return await asyncio.to_thread(self._client.health)
        except Exception as e:
            logger.error(f"InfluxDB health check failed: {e}")
            return False

influxdb_manager = InfluxDBManager()
```

**2.2.2 查询性能优化**

```python
# backend/src/services/monitoring_service.py
from influxdb_client import InfluxDBClient
from datetime import datetime, timedelta

class MonitoringService:

    async def get_system_performance(
        self,
        time_range: str = "24h",
        aggregate_window: str = "5m"  # 5分钟聚合窗口
    ) -> List[SystemPerformancePoint]:
        """查询系统性能历史 - InfluxDB 优化版"""

        # 解析时间范围
        range_map = {
            "1h": "1h",
            "6h": "6h",
            "24h": "24h",
            "7d": "7d",
            "30d": "30d"
        }

        # Flux 查询优化：
        # 1. 使用 range() 过滤时间（索引优化）
        # 2. 使用 filter() 精确匹配 measurement
        # 3. 使用 aggregateWindow() 降低数据点数量
        # 4. 使用 yield() 返回多个序列
        query = f'''
            from(bucket: "{settings.INFLUXDB_BUCKET}")
              |> range(start: -{range_map[time_range]})
              |> filter(fn: (r) => r["_measurement"] == "system_performance")
              |> filter(fn: (r) =>
                  r["_field"] == "cpu_percent" or
                  r["_field"] == "memory_percent" or
                  r["_field"] == "disk_percent"
              )
              |> aggregateWindow(
                  every: {aggregate_window},
                  fn: mean,  # 使用平均值聚合
                  createEmpty: false
              )
              |> yield(name: "mean")
        '''

        # 异步查询
        result = await asyncio.to_thread(
            influxdb_manager.query_api.query,
            query=query,
            org=settings.INFLUXDB_ORG
        )

        # 解析结果并转换为前端格式
        data_points = []
        for table in result:
            for record in table.records:
                data_points.append({
                    "timestamp": record.get_time().isoformat(),
                    "field": record.get_field(),
                    "value": round(record.get_value(), 2)
                })

        return self._transform_to_frontend_format(data_points)

    def _transform_to_frontend_format(self, raw_data: List) -> List[SystemPerformancePoint]:
        """转换 InfluxDB 数据为前端需要的格式"""
        # 按时间戳分组
        grouped = {}
        for point in raw_data:
            ts = point["timestamp"]
            if ts not in grouped:
                grouped[ts] = {"timestamp": ts}
            grouped[ts][point["field"]] = point["value"]

        return [
            SystemPerformancePoint(
                timestamp=ts,
                cpu=data.get("cpu_percent", 0),
                memory=data.get("memory_percent", 0),
                disk=data.get("disk_percent", 0)
            )
            for ts, data in sorted(grouped.items())
        ]
```

**2.2.4 查询缓存策略**

```python
# backend/src/services/monitoring_service.py
from functools import lru_cache
from datetime import datetime, timedelta

class MonitoringService:

    @lru_cache(maxsize=128)
    def _get_cache_key(self, time_range: str, query_time: str) -> str:
        """生成缓存键（按5分钟对齐）"""
        # 将查询时间向下对齐到5分钟边界
        dt = datetime.fromisoformat(query_time)
        aligned = dt.replace(second=0, microsecond=0)
        aligned = aligned - timedelta(minutes=aligned.minute % 5)
        return f"{time_range}:{aligned.isoformat()}"

    async def get_system_performance_cached(self, time_range: str):
        """带缓存的系统性能查询"""
        cache_key = self._get_cache_key(
            time_range,
            datetime.utcnow().isoformat()
        )

        # 尝试从 Redis 缓存读取
        cached = await redis_manager.get(f"monitoring:perf:{cache_key}")
        if cached:
            return json.loads(cached)

        # 缓存未命中，查询 InfluxDB
        data = await self.get_system_performance(time_range)

        # 写入缓存（5分钟过期）
        await redis_manager.setex(
            f"monitoring:perf:{cache_key}",
            300,  # 5分钟
            json.dumps(data, default=str)
        )

        return data
```

```typescript
'use client'

import { useState } from 'react'
import { toast } from 'react-hot-toast'

interface ReportExportButtonProps {
  timeRange: string
}

export function ReportExportButton({ timeRange }: ReportExportButtonProps) {
  const [isExporting, setIsExporting] = useState(false)
  const [format, setFormat] = useState<'pdf' | 'excel' | 'csv'>('pdf')

  const handleExport = async () => {
    setIsExporting(true)
    try {
      const response = await fetch('/api/v1/monitoring/reports/export', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('access_token')}`
        },
        body: JSON.stringify({
          format,
          time_range: timeRange,
          sections: ['stats', 'charts', 'alerts']
        })
      })

      if (!response.ok) throw new Error('导出失败')

      // 下载文件
      const blob = await response.blob()
      const url = window.URL.createObjectURL(blob)
      const a = document.createElement('a')
      a.href = url
      a.download = `监控报告_${new Date().toISOString()}.${format === 'excel' ? 'xlsx' : format}`
      a.click()
      window.URL.revokeObjectURL(url)

      toast.success('报告导出成功')
    } catch (error) {
      toast.error('报告导出失败')
    } finally {
      setIsExporting(false)
    }
  }

  return (
    <div className="flex gap-2">
      <select
        value={format}
        onChange={(e) => setFormat(e.target.value as any)}
        className="rounded-md border border-gray-300 px-3 py-1 text-sm"
      >
        <option value="pdf">PDF</option>
        <option value="excel">Excel</option>
        <option value="csv">CSV</option>
      </select>
      <button
        onClick={handleExport}
        disabled={isExporting}
        className="rounded-md bg-blue-600 px-4 py-1 text-sm font-medium text-white hover:bg-blue-700 disabled:bg-gray-400"
      >
        {isExporting ? '导出中...' : '📥 导出报告'}
      </button>
    </div>
  )
}
```
